   170→def _s(v) -> str:
   171→    """Clean string value — returns empty string for None/nan/empty."""
   172→    if v is None:
   173→        return ""
   174→    s = str(v).strip()
   175→    return "" if s in ("None", "nan", "0", "NaT", "#N/A", "#VALUE!") else s

   200→def _format_date_ddmmyyyy(raw_value) -> str:
   201→    """
   202→    Format a date value as DD-MM-YYYY, for AT_IncomingMonth.
   203→
   204→    Accepts a datetime/date object (openpyxl returns these for date-typed
   205→    cells) or a string in M/D/YYYY format (e.g. "9/23/2026" or "9/23/2026."
   206→    with a trailing period, as the Anta input file uses). Returns "" if the
   207→    value can't be parsed as a date — no guessing.
   208→    """
   209→    if raw_value is None:
   210→        return ""
   211→    if hasattr(raw_value, "day") and hasattr(raw_value, "month") and hasattr(raw_value, "year"):
   212→        return f"{raw_value.day:02d}-{raw_value.month:02d}-{raw_value.year}"
   213→    s = str(raw_value).strip().rstrip(".").strip()
   214→    m = re.match(r"^(\d{1,2})/(\d{1,2})/(\d{4})$", s)
   215→    if not m:
   216→        return ""
   217→    month, day, year = int(m.group(1)), int(m.group(2)), m.group(3)
   218→    return f"{day:02d}-{month:02d}-{year}"
   219→
   220→
   221→_MONTH_ABBR = {
   222→    1: "Jan", 2: "Feb", 3: "Mar", 4: "Apr", 5: "May", 6: "Jun",
   223→    7: "Jul", 8: "Aug", 9: "Sep", 10: "Oct", 11: "Nov", 12: "Dec",
   224→}
   225→
   226→
   227→def _format_date_ddmonyyyy(raw_value) -> str:
   228→    """
   229→    Format a date value as DD-Mon-YYYY (e.g. "23-Sep-2026"), for
   230→    AT_LaunchingDate.
   231→
   232→    Accepts a datetime/date object (openpyxl returns these for date-typed
   233→    cells) or a string in M/D/YYYY or M/D/YY format (e.g. "9/23/2026" or
   234→    "9/23/26", optionally with a trailing period). Returns "" if the value
   235→    can't be parsed as a date — no guessing.
   236→    """
   237→    if raw_value is None:
   238→        return ""
   239→    if hasattr(raw_value, "day") and hasattr(raw_value, "month") and hasattr(raw_value, "year"):
   240→        return f"{raw_value.day:02d}-{_MONTH_ABBR[raw_value.month]}-{raw_value.year}"
   241→    s = str(raw_value).strip().rstrip(".").strip()
   242→    m = re.match(r"^(\d{1,2})/(\d{1,2})/(\d{2}|\d{4})$", s)
   243→    if not m:
   244→        return ""
   245→    month, day, year = int(m.group(1)), int(m.group(2)), m.group(3)
   246→    if len(year) == 2:
   247→        year = f"20{year}"
   248→    if month not in _MONTH_ABBR:
   249→        return ""
   250→    return f"{day:02d}-{_MONTH_ABBR[month]}-{year}"

   256→
   257→def _val(parent: ET.Element, attr_id: str, text: str = "", lov_id: str = "") -> None:
   258→    if not text and not lov_id:
   259→        return
   260→    v = ET.SubElement(parent, f"{{{STIBO_NS}}}Value")
   261→    v.set("AttributeID", attr_id)
   262→    if lov_id:
   263→        v.set("ID", lov_id)
   264→    if text:
   265→        v.text = text
   266→
   267→
   268→def _val_text(parent: ET.Element, attr_id: str, text: str) -> None:
   269→    t = _s(text)
   270→    if t:
   271→        _val(parent, attr_id, text=t)
   272→
   273→
   274→def _val_lov(parent: ET.Element, attr_id: str, lov_id: str) -> None:
   275→    lid = _s(lov_id)
   276→    if lid:
   277→        _val(parent, attr_id, lov_id=lid)
   278→
   279→
   280→def _multival(parent: ET.Element, attr_id: str, id_val: str) -> None:
   281→    v = _s(id_val)
   282→    if not v:
   283→        return
   284→    mv = ET.SubElement(parent, f"{{{STIBO_NS}}}MultiValue")
   285→    mv.set("AttributeID", attr_id)
   286→    ET.SubElement(mv, f"{{{STIBO_NS}}}Value").set("ID", v)

   294→class MDDLoader:
   295→    """Loads the LOV tables needed to resolve AT_Brand / AT_BrandType /
   296→    AT_BrandStatus / AT_BrandGroup, plus Country (id -> name, used to key
   297→    the RNA lookup)."""

   299→    def __init__(self, path: Path):
   300→        self.path = path
   301→        self.lovs: dict[str, dict] = {}
   302→        self._load()

   304→    def _load(self) -> None:
   305→        log.info("[MDD] Loading: %s", self.path.name)
   306→        wb = openpyxl.load_workbook(self.path, read_only=True, data_only=True)
   307→        self._load_brand_lov(wb)
   308→        self._load_brand_type_lov(wb)
   309→        self._load_brand_status_lov(wb)
   310→        self._load_brand_group_lov(wb)
   311→        self._load_country_lov(wb)
   312→        self._load_ecom_ages_category_lov(wb)
   313→        self._load_sports_category_lov(wb)
   314→        wb.close()
   315→        log.info("[MDD] %d LOV types loaded", len(self.lovs))
   316→
   317→    def _load_brand_lov(self, wb) -> None:
   318→        """Brand LOV: col A = LOV ID, col B = display name."""
   319→        sheet = next(
   320→            (s for s in wb.sheetnames if "BRAND" in s.upper() and "LOV" in s.upper()
   321→             and "TYPE" not in s.upper() and "GROUP" not in s.upper() and "STATUS" not in s.upper()),
   322→            None,
   323→        )
   324→        if not sheet:
   325→            log.warning("[MDD] Brand LOV sheet not found")
   326→            return
   327→        lov: dict[str, str] = {}
   328→        for row in list(wb[sheet].iter_rows(values_only=True))[1:]:
   329→            if not row or len(row) < 2:
   330→                continue
   331→            lov_id  = str(row[0]).strip() if row[0] else ""
   332→            display = str(row[1]).strip() if row[1] else ""
   333→            if display and lov_id:
   334→                lov[display.upper()] = lov_id
   335→        self.lovs["AT_Brand"] = lov
   336→        log.info("[MDD] Brand LOV: %d entries", len(lov))
   337→

   595→# ATTRIBUTE MAPPING ENGINE  (drives everything — see module docstring)
   596→# ======================================================================

   598→def _parse_mapping_logic(raw_logic) -> dict[str, str]:
   599→    """
   600→    Parse Col K "Mapping Logic" text into {SOURCE_UPPER: resolved_id_or_text}.
   601→
   602→    Expected line format: "<source> : <target_id> - <target_label>"
   603→      e.g. "M : M - Male"   -> {"M": "M"}
   604→           "W : F - Female" -> {"W": "F"}
   605→    Lines that don't contain ":" (e.g. free-text notes like
   606→    "Mapping from Principal Data") are ignored — callers fall back to raw
   607→    passthrough when the resulting table is empty.
   608→    """
   609→    mapping: dict[str, str] = {}
   610→    text = _s(raw_logic)
   611→    if not text:
   612→        return mapping
   613→    for line in text.splitlines():
   614→        line = line.strip()
   615→        if not line or ":" not in line:
   616→            continue
   617→        src, _, tgt = line.partition(":")
   618→        src = src.strip()
   619→        tgt = tgt.strip()
   620→        if not src or not tgt:
   621→            continue
   622→        # "F - Female" -> take the id token before " - "
   623→        tgt_id = tgt.split(" - ", 1)[0].strip() or tgt
   624→        mapping[src.upper()] = tgt_id
   625→    return mapping

   628→class AttributeRule:
   629→    def __init__(self, name: str, attr_id: str, validation: str,
   630→                 mapping_type: str, source_column: str, mapping_logic_raw):
   631→        self.name          = name
   632→        self.attr_id       = attr_id
   633→        self.is_lov        = validation.strip().upper() == "LOV"
   634→        self.mapping_type  = mapping_type
   635→        self.source_column = source_column
   636→        self.mode          = "direct"   # "direct" | "mapped"
   637→        self.value_map: dict[str, str] = {}
   638→
   639→        mt = mapping_type.strip()
   640→        if mt == "Mapping from Principal Data":
   641→            self.mode = "mapped"
   642→            self.value_map = _parse_mapping_logic(mapping_logic_raw)
   643→
   644→    def resolve(self, raw_value: str) -> str | None:
   645→        """Return the value to write for this attribute, or None to skip it."""
   646→        raw = _s(raw_value)
   647→        if not raw:
   648→            return None
   649→        if self.mode == "direct" or not self.value_map:
   650→            return raw
   651→        # "mapped" mode with a real lookup table — no default guess.
   652→        resolved = self.value_map.get(raw.upper())
   653→        if resolved is None:
   654→            log.info(
   655→                "[AttrMap] '%s' (%s): raw value '%s' not in Col K mapping table "
   656→                "%s — attribute skipped for this row",
   657→                self.name, self.attr_id, raw, sorted(self.value_map.keys()),
   658→            )
   659→        return resolved
