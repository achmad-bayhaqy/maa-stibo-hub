   146→def _parse_country_code_from_file_or_args(filename: str, cli_country: str = "") -> str:
   147→    # 1. First check if filename has an explicit country token (e.g. PH, KH, TH, MY, SG, VN, ID)
   148→    stem_upper = Path(filename).stem.upper().replace("_", " ").replace("-", " ")
   149→    tokens = stem_upper.split()
   150→    for token in tokens:
   151→        if token in _COUNTRY_NAME_TO_CODE and token != "SP":
   152→            return _COUNTRY_NAME_TO_CODE[token]
   153→
   154→    # 2. Fallback to CLI argument if provided
   155→    cli_c = (cli_country or "").strip().upper()
   156→    if cli_c:
   157→        return cli_c
   158→            
   159→    return "ID"

   295→class MDDLoader:
   296→    """Loads Core Attributes + LOVs from the MDD Excel."""
   297→
   298→    def __init__(self, path: Path):
   299→        self.path       = path
   300→        self.attributes: dict[str, dict] = {}
   301→        self.lovs:       dict[str, dict] = {}
   302→        self._load()
   303→
   304→    def _load(self):
   305→        log.info("[MDD] Loading: %s", self.path.name)
   306→        wb  = openpyxl.load_workbook(self.path, read_only=True, data_only=True)
   307→        ws  = wb["Core Attributes"]
   308→        rows = list(ws.iter_rows(values_only=True))
   309→        hdr  = rows[1]
   310→
   311→        col: dict[str, int] = {}
   312→        for i, h in enumerate(hdr):
   313→            if h:
   314→                col[str(h).split("\n")[0].strip()] = i
   315→                col[str(h).strip()] = i
   316→
   317→        for row in rows[2:]:
   318→            aid = row[7] if len(row) > 7 else None
   319→            if not aid:
   320→                continue
   321→            aid = str(aid).strip()
   322→
   323→            def _v(key):
   324→                idx = col.get(key)
   325→                if idx is None or idx >= len(row):
   326→                    return None
   327→                v = row[idx]
   328→                return str(v).strip() if v else None
   329→
   330→            self.attributes[aid] = {
   331→                "id":           aid,
   332→                "name":         _v("PIM Attribute Name"),
   333→                "source_name":  _v("Source Attribute Name"),
   334→                "cardinality":  _v("Cardinality"),
   335→                "validation":   _v("Validation Base Type"),
   336→                "multi_valued": _v("Multi Valued"),
   337→                "lov_name":     _v("Name of LOV"),
   338→                "max_chars":    _v("Max Characters"),
   339→                "group":        _v("PIM Attribute Group"),
   340→            }
   341→
   342→        self._load_simple_lovs(wb)
   343→        self._load_named_lov_sheets(wb)
   344→        self._load_age_lov(wb)
   345→        self._load_gender_lov(wb)
   346→        wb.close()
   347→        log.info("[MDD] %d attributes | %d LOVs", len(self.attributes), len(self.lovs))
   348→
   349→    def _load_simple_lovs(self, wb):
   350→        if "Simple LOVs" not in wb.sheetnames:
   351→            return
   352→        for row in list(wb["Simple LOVs"].iter_rows(values_only=True))[1:]:
   353→            lov_name, _, val_name, val_id = (row + (None,) * 4)[:4]
   354→            if lov_name and val_name:
   355→                self.lovs.setdefault(str(lov_name).strip(), {})[
   356→                    str(val_name).strip()
   357→                ] = str(val_id).strip() if val_id else str(val_name).strip()
   358→
   359→    def _load_age_lov(self, wb):
   360→        """Load Age LOV sheet: col A = display value, col B = LOV ID. Also loads SAPAge and BYAge."""
   361→        sheet_name = next((s for s in wb.sheetnames if "AGE" in s.upper() and "LOV" in s.upper()), None)
   362→        if not sheet_name:
   363→            log.warning("[MDD] Age LOV sheet not found")
   364→            return
   365→
   366→        rows = list(wb[sheet_name].iter_rows(values_only=True))
   367→        if len(rows) < 2:
   368→            return
   369→
   370→        sap_age_map = {}
   371→        by_age_map  = {}
   372→        age_lov_map = {}  # col A (display) → col B (LOV ID)
   373→        by_age_lov_map = {}  # col G (BY Age display) → col F (LOV ID)
   374→
   375→        for row in rows[1:]:
   376→            if not row or not row[0]:
   377→                continue
   378→            age_display  = str(row[0]).strip()
   379→            age_lov_id   = str(row[1]).strip() if len(row) > 1 and row[1] else ""
   380→            sap_age_code = str(row[2]).strip() if len(row) > 2 and row[2] else ""
   381→            by_age_val   = str(row[5]).strip() if len(row) > 5 and row[5] else ""
   382→            by_age_disp  = str(row[6]).strip() if len(row) > 6 and row[6] else ""  # col G
   383→            by_age_id    = str(row[5]).strip() if len(row) > 5 and row[5] else ""  # col F
   384→
   385→            if age_display and age_lov_id:
   386→                age_lov_map[age_display] = age_lov_id
   387→            if age_display and sap_age_code:
   388→                sap_age_map[age_display] = sap_age_code
   389→            if age_display and by_age_val:
   390→                by_age_map[age_display] = by_age_val
   391→            if by_age_disp and by_age_id:
   392→                by_age_lov_map[by_age_disp] = by_age_id
   393→
   394→        self.lovs["AgeLOV"]    = age_lov_map
   395→        self.lovs["ByAgeLOV"]  = by_age_lov_map
   396→        self.lovs["SAPAge"]    = sap_age_map
   397→        self.lovs["BYAge"]     = by_age_map
   398→        log.info("[MDD] Age LOV loaded — AgeLOV: %d, ByAgeLOV: %d, SAPAge: %d, BYAge: %d",
   399→                 len(age_lov_map), len(by_age_lov_map), len(sap_age_map), len(by_age_map))
   400→
   401→    def _load_gender_lov(self, wb):
   402→        """Load Gender LOV sheet: col A = display value, col C = LOV ID."""
   403→        sheet_name = next((s for s in wb.sheetnames if "GENDER" in s.upper() and "LOV" in s.upper()), None)
   404→        if not sheet_name:
   405→            log.warning("[MDD] Gender LOV sheet not found")
   406→            return
   407→        rows = list(wb[sheet_name].iter_rows(values_only=True))
   408→        gender_lov: dict[str, str] = {}
   409→        for row in rows[1:]:
   410→            if not row or not row[0]:
   411→                continue
   412→            display = str(row[0]).strip()
   413→            lov_id  = str(row[2]).strip() if len(row) > 2 and row[2] else ""
   414→            if display and lov_id:
   415→                gender_lov[display] = lov_id
   416→        self.lovs["GenderLOV"] = gender_lov
   417→        log.info("[MDD] Gender LOV loaded — %d entries", len(gender_lov))
   418→
   419→    def _load_named_lov_sheets(self, wb):
   420→        for sn in wb.sheetnames:
   421→            if "LOV" not in sn.upper():
   422→                continue
   423→            rows = list(wb[sn].iter_rows(values_only=True))
   424→            if len(rows) < 2:
   425→                continue
   426→            display = sn.replace(" LOV", "").replace("LOV ", "").strip()
   427→            
   428→            for row in rows[1:]:
   429→                if not row or len(row) < 2:
   430→                    continue
   431→                name, code = row[0], row[1]
   432→                if name:
   433→                    self.lovs.setdefault(display, {})[str(name).strip()] = (
   434→                        str(code).strip() if code else str(name).strip()
   435→                    )

   438→class EllesseMDMappingLoader:
   439→    """
   440→    Loads the 'Lotto MD Mapping' tab from the Brand Mapping workbook.
   441→    Col G (index 6)  = source value from input file.
   442→    Col H (index 7)  = SAP Age LOV display value    → used for AT_SAPAge.
   443→    Col I (index 8)  = SAP Gender LOV display value → used for AT_Gender.
   444→    Col J (index 9)  = BY Age LOV display value     → used for AT_BYAge.
   445→    Col K (index 10) = BY Gender LOV display value  → used for AT_BYGender.
   446→    """
   447→
   448→    def __init__(self, path: Path):
   449→        self.path          = path
   450→        self.col_g_to_h: dict[str, str] = {}  # {source_upper: sap_age_display}
   451→        self.col_g_to_i: dict[str, str] = {}  # {source_upper: sap_gender_display}
   452→        self.col_g_to_j: dict[str, str] = {}  # {source_upper: by_age_display}
   453→        self.col_g_to_k: dict[str, str] = {}  # {source_upper: by_gender_display}
   454→        self._load()
   455→
   456→    def _load(self):
   457→        log.info("[MDMapping] Loading from: %s", self.path.name)
   458→        wb = openpyxl.load_workbook(self.path, read_only=True, data_only=True)
   459→        sheet_name = "Ellesse MD Mapping"
   460→        if sheet_name not in wb.sheetnames:
   461→            log.warning("[MDMapping] '%s' sheet not found", sheet_name)
   462→            wb.close()
   463→            return
   464→
   465→        ws   = wb[sheet_name]
   466→        rows = list(ws.iter_rows(values_only=True))
   467→        count = 0
   468→        for row in rows:
   469→            if not row or len(row) <= 8:
   470→                continue
   471→            src   = row[6]                               # Col G
   472→            sap_a = row[7]                               # Col H — SAP Age
   473→            sap_g = row[8]                               # Col I — SAP Gender
   474→            by_a  = row[9]  if len(row) > 9  else None  # Col J — BY Age
   475→            by_g  = row[10] if len(row) > 10 else None  # Col K — BY Gender
   476→            if not src:
   477→                continue
   478→            src_str = str(src).strip().upper()
   479→            if sap_a: self.col_g_to_h[src_str] = str(sap_a).strip()
   480→            if sap_g: self.col_g_to_i[src_str] = str(sap_g).strip()
   481→            if by_a:  self.col_g_to_j[src_str] = str(by_a).strip()
   482→            if by_g:  self.col_g_to_k[src_str] = str(by_g).strip()
   483→            if any([sap_a, sap_g, by_a, by_g]):
   484→                count += 1
   485→
   486→        wb.close()
   487→        log.info("[MDMapping] %d mappings loaded from '%s'", count, sheet_name)

   565→class RNALoader:
   566→    """
   567→    Loads the 'Source Mapping Related RNA' tab from the Attributes List workbook.
   568→    Maps (Country Name, Company Code, SBU, Brand Code) → Brand Type + Brand Category.
   569→    """
   570→
   571→    RNA_SHEET_KEYWORDS = ["RNA", "SOURCE MAPPING"]
   572→
   573→    @staticmethod
   574→    def _norm_country(v: str) -> str:
   575→        return (v or "").strip().upper()
   576→
   577→    @staticmethod
   578→    def _norm_comp_code(v: str) -> str:
   579→        s = (v or "").strip()
   580→        if not s:
   581→            return ""
   582→        # ── FIX: strip .0 suffix (e.g. "888.0" → "888") ─────────
   583→        if s.endswith(".0"):
   584→            s = s[:-2]
   585→        s2 = s.lstrip("0")
   586→        return s2 if s2 else "0"
   587→
   588→    def __init__(self, path: Path):
   589→        self.path   = path
   590→        self.lookup: dict[tuple, dict] = {}
   591→        self._load()
   592→
   593→    def _load(self):
   594→        log.info("[RNA] Loading from: %s", self.path.name)
   595→        wb = openpyxl.load_workbook(self.path, read_only=True, data_only=True)
   596→
   597→        sheet_name = next(
   598→            (s for s in wb.sheetnames if any(kw in s.upper() for kw in self.RNA_SHEET_KEYWORDS)),
   599→            None,
   600→        )
   601→        if not sheet_name:
   602→            log.warning("[RNA] 'Source Mapping Related RNA' tab not found in %s", self.path.name)
   603→            wb.close()
   604→            return
   605→
   606→        log.info("[RNA] Using sheet: '%s'", sheet_name)
   607→        rows = list(wb[sheet_name].iter_rows(values_only=True))
   608→
   609→        hdr_idx = next(
   610→            (i for i, r in enumerate(rows)
   611→             if any(isinstance(v, str) and "COUNTRY" in v.upper() for v in r if v)),
   612→            None,
   613→        )
   614→        if hdr_idx is None:
   615→            log.warning("[RNA] Cannot find header row in sheet '%s'", sheet_name)
   616→            wb.close()
   617→            return
   618→
   619→        hdr = rows[hdr_idx]
   620→        col = {str(h).strip().upper(): i for i, h in enumerate(hdr) if h}
   621→
   622→        def _find(*candidates) -> int | None:
   623→            for c in candidates:
   624→                if c.upper() in col:
   625→                    return col[c.upper()]
   626→            return None
   627→
   628→        c_country = _find("COUNTRY", "COUNTRY NAME")
   629→        c_comp    = _find("COMPANY CODE", "COMP CODE", "COMPCODE", "COMP_CODE")
   630→        c_sbu     = _find("SBU")
   631→        c_bcode   = _find("BRANDCODE", "BRAND CODE", "REPORTING BRAND CODE MAPPED")
   632→        c_btype   = _find("BRANDTYPE_DETAIL", "BRAND TYPE", "AT_BRANDTYPE", "BRANDTYPE")
   633→        c_bcat    = _find("BRANDCATEGORY", "BRAND CATEGORY", "AT_BRANDCATEGORY", "BRANDCATEGORY")
   634→
   635→        missing = [nm for nm, idx in [
   636→            ("Country", c_country), ("CompCode", c_comp), ("SBU", c_sbu),
   637→            ("BrandCode", c_bcode), ("BrandType", c_btype), ("BrandCategory", c_bcat),
   638→        ] if idx is None]
   639→        if missing:
   640→            log.warning("[RNA] Missing columns in '%s': %s", sheet_name, missing)
   641→
   642→        def _cell(row, idx):
   643→            if idx is None or idx >= len(row) or row[idx] is None:
   644→                return ""
   645→            return str(row[idx]).strip()
   646→
   647→        count = 0
   648→        for row in rows[hdr_idx + 1:]:
   649→            country = _cell(row, c_country)
   650→            if not country:
   651→                continue
   652→            key = (
   653→                self._norm_country(country),
   654→                self._norm_comp_code(_cell(row, c_comp)),
   655→                _cell(row, c_sbu).upper(),
   656→                _cell(row, c_bcode).upper(),
   657→            )
   658→            self.lookup[key] = {
   659→                "brand_type":     _cell(row, c_btype),
   660→                "brand_category": _cell(row, c_bcat),
   661→            }
   662→            count += 1
   663→
   664→        wb.close()
   665→        log.info("[RNA] %d entries loaded from '%s'", count, sheet_name)
   666→
   667→    def get(self, country_name: str, comp_code: str, sbu: str, brand_code: str) -> dict:
   668→        """Return {brand_type, brand_category} for the given keys, or empty strings."""
   669→        key = (
   670→            self._norm_country(country_name),
   671→            self._norm_comp_code(comp_code),
   672→            (sbu        or "").strip().upper(),
   673→            (brand_code or "").strip().upper(),
   674→        )
   675→        return self.lookup.get(key, {"brand_type": "", "brand_category": ""})
   676→
   677→    def get_fuzzy(self, country_name: str, sbu: str, brand_code: str) -> dict:
   678→        """Match on country + brand only — ignores comp_code."""
   679→        c = self._norm_country(country_name)
   680→        b = (brand_code or "").strip().upper()
   681→        s = (sbu or "").strip().upper()
   682→        # pass 1: country + sbu + brand
   683→        for (kc, _, ks, kb), val in self.lookup.items():
   684→            if kc == c and ks == s and kb == b:
   685→                return val
   686→        # pass 2: country + brand only (first match)
   687→        for (kc, _, ks, kb), val in self.lookup.items():
   688→            if kc == c and kb == b:
   689→                return val
   690→        return {"brand_type": "", "brand_category": ""}

   880→    """country_code (e.g. 'ID', 'KH') → Retail Price Currency LOV ID (e.g. 'IDR', 'USD')."""
   881→    fallback_map = {
   882→        "ID": "IDR", "PH": "PHP", "TH": "THB",
   883→        "SG": "SGD", "MY": "MYR", "VN": "VND", "KH": "USD"
   884→    }

   886→    if not code_upper:
   887→        return ""
   888→
   889→    if mdd:
   890→        country_name = mdd.get("country_id_to_name", {}).get(code_upper, "")
   891→        retail_currency_lov = mdd.get("retail_price_currency_lov", {})
   892→
   893→        if country_name and retail_currency_lov:
   894→            country_name_upper = country_name.strip().upper()
   895→
   896→            # Special case mapping for countries like Cambodia where country name != currency name
   897→            if code_upper == "KH" or "CAMBODIA" in country_name_upper:
   898→                for currency_name, c_code in retail_currency_lov.items():
   899→                    if "UNITED STATES" in currency_name or c_code == "USD":
   900→                        return c_code
   901→                return "USD"
   902→
   903→            if country_name_upper in retail_currency_lov:
   904→                return retail_currency_lov[country_name_upper]
   905→
   906→            country_root = country_name_upper.rstrip("S").rstrip("N")
   907→            for lov_entry, currency in retail_currency_lov.items():
   908→                lov_entry_upper = lov_entry.upper()

   910→                    country_root in lov_entry_upper or

   913→
   914→    # Fallback if MDD lookup didn't yield a match
   915→    return fallback_map.get(code_upper, "")

   918→def _fmt_date(v) -> str:
   919→    """Format a date to dd-Mon-YYYY lowercase."""
   920→    if isinstance(v, datetime):
   921→        return v.strftime("%d-%b-%Y").lower()
   922→    if hasattr(v, "strftime"):
   923→        return v.strftime("%d-%b-%Y").lower()
   924→    raw = _s(v)
   925→    if not raw:
   926→        return raw
   927→    for fmt in ("%Y-%m-%d", "%d/%m/%Y", "%m/%d/%Y", "%d.%m.%Y"):
   928→        try:
   929→            return datetime.strptime(raw, fmt).strftime("%d-%b-%Y").lower()
   930→        except (ValueError, TypeError):
   931→            pass
   932→    return raw

   935→def _size_to_sap_code(size_raw: str) -> str:
   936→    """
   937→    Convert a single LOTTO size string to a 3-char SAP size code.
   938→    Falls back to a zero-padded or truncated version of the raw value.
   939→    """
   940→    key = size_raw.strip().upper()
   941→    if key in ELLESSE_SIZE_TO_SAP:
   942→        return ELLESSE_SIZE_TO_SAP[key]
   943→    
   944→    # Try numeric conversion (e.g., "37" → "037")
   945→    if key.isdigit():
   946→        return key.zfill(3)
   947→    
   948→    # Generic fallback: strip non-alphanumeric, take first 3 chars, pad to 3
   949→    cleaned = re.sub(r"[^A-Z0-9]", "", key)[:3].ljust(3, "X")
   950→    return cleaned

   953→def _parse_season(season: str) -> tuple[str, str]:
   954→    """
   955→    Split a season token into (sea_code, sea_year).
   956→    
   957→    Examples:
   958→        SS26  → ("SS", "2026")
   959→        FW27  → ("FW", "2027")
   960→        SP2028 → ("SP", "2028")
   961→    """
   962→    s = (season or "").strip().upper()
   963→    if len(s) < 3:
   964→        return s, ""
   965→    
   966→    # Match 2-letter code + 2 or 4 digit year
   967→    match = re.match(r"^([A-Z]{2})(\d{2,4})$", s)
   968→    if match:
   969→        code = match.group(1)
   970→        year_part = match.group(2)
   971→        # Convert 2-digit year to 4-digit
   972→        if len(year_part) == 2:
   973→            year = f"20{year_part}"
   974→        else:
   975→            year = year_part
   976→        return code, year
   977→    
   978→    return s[:2], ""
