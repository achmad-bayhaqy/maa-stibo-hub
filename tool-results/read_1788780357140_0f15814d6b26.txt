   294→   294→
   295→   295→class MDDLoader:
   296→   296→    """Loads Core Attributes + LOVs from the MDD Excel."""
   297→   297→
   298→   298→    def __init__(self, path: Path):
   299→   299→        self.path       = path
   300→   300→        self.attributes: dict[str, dict] = {}
   301→   301→        self.lovs:       dict[str, dict] = {}
   302→   302→        self._load()
   303→   303→
   304→   304→    def _load(self):
   305→   305→        log.info("[MDD] Loading: %s", self.path.name)
   306→   306→        wb  = openpyxl.load_workbook(self.path, read_only=True, data_only=True)
   307→   307→        ws  = wb["Core Attributes"]
   308→   308→        rows = list(ws.iter_rows(values_only=True))
   309→   309→        hdr  = rows[1]
   310→   310→
   311→   311→        col: dict[str, int] = {}
   312→   312→        for i, h in enumerate(hdr):
   313→   313→            if h:
   314→   314→                col[str(h).split("\n")[0].strip()] = i
   315→   315→                col[str(h).strip()] = i
   316→   316→
   317→   317→        for row in rows[2:]:
   318→   318→            aid = row[7] if len(row) > 7 else None
   319→   319→            if not aid:
   320→   320→                continue
   321→   321→            aid = str(aid).strip()
   322→   322→
   323→   323→            def _v(key):
   324→   324→                idx = col.get(key)
   325→   325→                if idx is None or idx >= len(row):
   326→   326→                    return None
   327→   327→                v = row[idx]
   328→   328→                return str(v).strip() if v else None
   329→   329→
   330→   330→            self.attributes[aid] = {
   331→   331→                "id":           aid,
   332→   332→                "name":         _v("PIM Attribute Name"),
   333→   333→                "source_name":  _v("Source Attribute Name"),
   334→   334→                "cardinality":  _v("Cardinality"),
   335→   335→                "validation":   _v("Validation Base Type"),
   336→   336→                "multi_valued": _v("Multi Valued"),
   337→   337→                "lov_name":     _v("Name of LOV"),
   338→   338→                "max_chars":    _v("Max Characters"),
   339→   339→                "group":        _v("PIM Attribute Group"),
   340→   340→            }
   341→   341→
   342→   342→        self._load_simple_lovs(wb)
   343→   343→        self._load_named_lov_sheets(wb)
   344→   344→        self._load_age_lov(wb)
   345→   345→        self._load_gender_lov(wb)
   346→   346→        wb.close()
   347→   347→        log.info("[MDD] %d attributes | %d LOVs", len(self.attributes), len(self.lovs))
   348→   348→
   349→   349→    def _load_simple_lovs(self, wb):
   350→   350→        if "Simple LOVs" not in wb.sheetnames:
   351→   351→            return
   352→   352→        for row in list(wb["Simple LOVs"].iter_rows(values_only=True))[1:]:
   353→   353→            lov_name, _, val_name, val_id = (row + (None,) * 4)[:4]
   354→   354→            if lov_name and val_name:
   355→   355→                self.lovs.setdefault(str(lov_name).strip(), {})[
   356→   356→                    str(val_name).strip()
   357→   357→                ] = str(val_id).strip() if val_id else str(val_name).strip()
   358→   358→
   359→   359→    def _load_age_lov(self, wb):
   360→   360→        """Load Age LOV sheet: col A = display value, col B = LOV ID. Also loads SAPAge and BYAge."""
   361→   361→        sheet_name = next((s for s in wb.sheetnames if "AGE" in s.upper() and "LOV" in s.upper()), None)
   362→   362→        if not sheet_name:
   363→   363→            log.warning("[MDD] Age LOV sheet not found")
   364→   364→            return
   365→   365→
   366→   366→        rows = list(wb[sheet_name].iter_rows(values_only=True))
   367→   367→        if len(rows) < 2:
   368→   368→            return
   369→   369→
   370→   370→        sap_age_map = {}
   371→   371→        by_age_map  = {}
   372→   372→        age_lov_map = {}  # col A (display) → col B (LOV ID)
   373→   373→        by_age_lov_map = {}  # col G (BY Age display) → col F (LOV ID)
   374→   374→
   375→   375→        for row in rows[1:]:
   376→   376→            if not row or not row[0]:
   377→   377→                continue
   378→   378→            age_display  = str(row[0]).strip()
   379→   379→            age_lov_id   = str(row[1]).strip() if len(row) > 1 and row[1] else ""
   380→   380→            sap_age_code = str(row[2]).strip() if len(row) > 2 and row[2] else ""
   381→   381→            by_age_val   = str(row[5]).strip() if len(row) > 5 and row[5] else ""
   382→   382→            by_age_disp  = str(row[6]).strip() if len(row) > 6 and row[6] else ""  # col G
   383→   383→            by_age_id    = str(row[5]).strip() if len(row) > 5 and row[5] else ""  # col F
   384→   384→
   385→   385→            if age_display and age_lov_id:
   386→   386→                age_lov_map[age_display] = age_lov_id
   387→   387→            if age_display and sap_age_code:
   388→   388→                sap_age_map[age_display] = sap_age_code
   389→   389→            if age_display and by_age_val:
   390→   390→                by_age_map[age_display] = by_age_val
   391→   391→            if by_age_disp and by_age_id:
   392→   392→                by_age_lov_map[by_age_disp] = by_age_id
   393→   393→
   394→   394→        self.lovs["AgeLOV"]    = age_lov_map
   395→   395→        self.lovs["ByAgeLOV"]  = by_age_lov_map
   396→   396→        self.lovs["SAPAge"]    = sap_age_map
   397→   397→        self.lovs["BYAge"]     = by_age_map
   398→   398→        log.info("[MDD] Age LOV loaded — AgeLOV: %d, ByAgeLOV: %d, SAPAge: %d, BYAge: %d",
   399→   399→                 len(age_lov_map), len(by_age_lov_map), len(sap_age_map), len(by_age_map))
   400→   400→
   401→   401→    def _load_gender_lov(self, wb):
   402→   402→        """Load Gender LOV sheet: col A = display value, col C = LOV ID."""
   403→   403→        sheet_name = next((s for s in wb.sheetnames if "GENDER" in s.upper() and "LOV" in s.upper()), None)
   404→   404→        if not sheet_name:
   405→   405→            log.warning("[MDD] Gender LOV sheet not found")
   406→   406→            return
   407→   407→        rows = list(wb[sheet_name].iter_rows(values_only=True))
   408→   408→        gender_lov: dict[str, str] = {}
   409→   409→        for row in rows[1:]:
   410→   410→            if not row or not row[0]:
   411→   411→                continue
   412→   412→            display = str(row[0]).strip()
   413→   413→            lov_id  = str(row[2]).strip() if len(row) > 2 and row[2] else ""
   414→   414→            if display and lov_id:
   415→   415→                gender_lov[display] = lov_id
   416→   416→        self.lovs["GenderLOV"] = gender_lov
   417→   417→        log.info("[MDD] Gender LOV loaded — %d entries", len(gender_lov))
   418→   418→
   419→   419→    def _load_named_lov_sheets(self, wb):

   565→   565→class RNALoader:
   566→   566→    """

   606→   606→        log.info("[RNA] Using sheet: '%s'", sheet_name)

   610→   610→            (i for i, r in enumerate(rows)
   611→   611→             if any(isinstance(v, str) and "COUNTRY" in v.upper() for v in r if v)),

   622→   622→        def _find(*candidates) -> int | None:
   623→   623→            for c in candidates:
   624→   624→                if c.upper() in col:

   670→   670→            self._norm_country(country_name),
   671→   671→            self._norm_comp_code(comp_code),
   672→   672→            (sbu        or "").strip().upper(),
   673→   673→            (brand_code or "").strip().upper(),
   674→   674→        )
   675→   675→        return self.lookup.get(key, {"brand_type": "", "brand_category": ""})

   677→   677→    def get_fuzzy(self, country_name: str, sbu: str, brand_code: str) -> dict:
   678→   678→        """Match on country + brand only — ignores comp_code."""
   679→   679→        c = self._norm_country(country_name)
   680→   680→        b = (brand_code or "").strip().upper()
   681→   681→        s = (sbu or "").strip().upper()
   682→   682→        # pass 1: country + sbu + brand
   683→   683→        for (kc, _, ks, kb), val in self.lookup.items():
   684→   684→            if kc == c and ks == s and kb == b:
   685→   685→                return val
   686→   686→        # pass 2: country + brand only (first match)
   687→   687→        for (kc, _, ks, kb), val in self.lookup.items():
   688→   688→            if kc == c and kb == b:
   689→   689→                return val
   690→   690→        return {"brand_type": "", "brand_category": ""}

   953→   953→def _parse_season(season: str) -> tuple[str, str]:
   954→   954→    """
   955→   955→    Split a season token into (sea_code, sea_year).
   956→   956→    
   957→   957→    Examples:

   964→   964→        return s, ""

   968→   968→    if match:
   969→   969→        code = match.group(1)
   970→   970→        year_part = match.group(2)
   971→   971→        # Convert 2-digit year to 4-digit
   972→   972→        if len(year_part) == 2:
   973→   973→            year = f"20{year_part}"
   974→   974→        else:
   975→   975→            year = year_part
